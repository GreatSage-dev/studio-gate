def evaluate_policy(rolling_burn, req_cost, cap):
    projected = round(rolling_burn + req_cost, 2)
    if projected > cap:
        return {"verdict": "BLOCKED", "overage": projected - cap}
    return {"verdict": "APPROVED", "overage": 0.0}
